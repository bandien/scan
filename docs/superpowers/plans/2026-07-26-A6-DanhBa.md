## Task 2 — JS: navigateToContacts() + renderContactsScreen() [~10 phút]

**GREEN — JS**:
```js
function navigateToContacts() {
  showScreen('contacts');
  renderContactsScreen();
}

let contactSearchQuery = '';

function renderContactsScreen() {
  const q = contactSearchQuery.trim().toLowerCase();
  let list = [...staffDirectory];
  if (q) {
    list = list.filter(s => {
      const name = (s.fullName || s.name || '').toLowerCase();
      const dept = (s.dept || '').toLowerCase();
      const phone = (s.phone || s.mobile || '').toLowerCase();
      return name.includes(q) || dept.includes(q) || phone.includes(q);
    });
  }

  if (list.length === 0) {
    document.getElementById('contactsBody').innerHTML =
      `<div class="nk-empty"><div class="nk-empty__icon">👤</div>
       ${q ? 'Không tìm thấy kết quả.' : 'Danh bạ trống.'}</div>`;
    return;
  }

  // Group theo dept
  const groups = {};
  list.forEach(s => {
    const dept = s.dept || 'Khác';
    if (!groups[dept]) groups[dept] = [];
    groups[dept].push(s);
  });

  const html = Object.entries(groups).sort((a,b) => a[0].localeCompare(b[0], 'vi')).map(([dept, members]) => `
    <div class="nk-contact-group">
      <div class="nk-contact-group__label">${escapeHtml(dept)}</div>
      ${members.map(s => {
        const name = s.fullName || s.name || '?';
        const short = getShortName(name);
        const phone = s.phone || s.mobile || '';
        const initials = name.charAt(0).toUpperCase();
        return `
          <div class="nk-contact-card">
            <div class="nk-contact-card__avatar">${initials}</div>
            <div class="nk-contact-card__info">
              <div class="nk-contact-card__name">${escapeHtml(name)}</div>
              <div class="nk-contact-card__meta">${escapeHtml(short)}${phone ? ' · ' + escapeHtml(phone) : ''}</div>
            </div>
            ${phone ? `<a class="nk-contact-card__call" href="tel:${escapeAttr(phone)}">📞</a>` : ''}
          </div>`;
      }).join('')}
    </div>`).join('');

  document.getElementById('contactsBody').innerHTML = html;
}
```
